cmake_minimum_required(VERSION 3.20)
project(reality_bridge_synapse VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(reality_bridge_synapse STATIC ../src/synapse.cpp)
target_include_directories(reality_bridge_synapse PUBLIC ../include)
add_executable(synapse_conformance ../examples/synapse_conformance.cpp)
target_link_libraries(synapse_conformance PRIVATE reality_bridge_synapse)
enable_testing()
add_executable(synapse_test ../tests/test_synapse.cpp)
target_link_libraries(synapse_test PRIVATE reality_bridge_synapse)
add_test(NAME synapse_test COMMAND synapse_test)
add_test(NAME synapse_conformance COMMAND synapse_conformance)

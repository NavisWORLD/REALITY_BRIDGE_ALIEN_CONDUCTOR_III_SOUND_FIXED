cmake_minimum_required(VERSION 3.20)
project(reality_bridge_dsp VERSION 0.1.0 LANGUAGES CXX)
option(RB_BUILD_SHARED "Build shared C ABI library" ON)
option(RB_BUILD_TESTS "Build tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(reality_bridge STATIC src/dsp.cpp)
target_include_directories(reality_bridge PUBLIC include)
if(RB_BUILD_SHARED)
  add_library(reality_bridge_c SHARED src/dsp.cpp src/c_api.cpp)
  target_compile_definitions(reality_bridge_c PRIVATE RB_BUILD_SHARED=1)
  target_include_directories(reality_bridge_c PUBLIC include)
endif()
add_executable(reality_bridge_demo examples/demo.cpp)
target_link_libraries(reality_bridge_demo PRIVATE reality_bridge)
if(RB_BUILD_TESTS)
  enable_testing()
  add_executable(reality_bridge_dsp_test tests/test_dsp.cpp)
  target_link_libraries(reality_bridge_dsp_test PRIVATE reality_bridge)
  add_test(NAME reality_bridge_dsp_test COMMAND reality_bridge_dsp_test)
endif()
